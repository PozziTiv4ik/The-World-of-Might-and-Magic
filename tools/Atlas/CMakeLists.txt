cmake_minimum_required(VERSION 3.20)
project(Atlas LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "Atlas uses the Windows desktop APIs")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(ATLAS_SOURCES
  src/platform.cpp src/pdn.cpp src/model.cpp src/render.cpp src/responsive.cpp src/land.cpp src/borders.cpp src/camera_test.cpp src/political.cpp src/controls.cpp src/boundary_render.cpp src/editor_controls.cpp
  src/history_delta.cpp src/history_store.cpp src/history_cli.cpp src/history_tests.cpp src/history_view.cpp src/timeline_model.cpp src/workspace.cpp src/workspace_tests.cpp src/editor_scenario.cpp src/recovery.cpp src/dialogs.cpp src/app.cpp src/ui.cpp src/actions.cpp src/cli.cpp src/tests.cpp src/performance.cpp src/preview.cpp)
add_library(atlas_objects OBJECT ${ATLAS_SOURCES})
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/symbols.json" ATLAS_SYMBOLS_JSON)
configure_file(src/symbols_data.hpp.in generated/symbols_data.hpp @ONLY)
target_include_directories(atlas_objects PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(atlas_objects PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00 NTDDI_VERSION=0x0A000003)
if(MSVC)
  target_compile_options(atlas_objects PRIVATE /utf-8 /W4 /EHsc)
endif()
set_source_files_properties(src/atlas.rc PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/atlas.manifest")
add_executable(Atlas WIN32 src/main.cpp src/atlas.rc $<TARGET_OBJECTS:atlas_objects>)
add_executable(Atlas.Cli src/cli_main.cpp src/atlas.rc $<TARGET_OBJECTS:atlas_objects>)
foreach(target Atlas Atlas.Cli)
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE)
  if(MSVC)
    target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/utf-8> $<$<COMPILE_LANGUAGE:CXX>:/W4> $<$<COMPILE_LANGUAGE:CXX>:/EHsc>)
    target_link_options(${target} PRIVATE /MANIFEST:NO)
  endif()
  target_link_libraries(${target} PRIVATE d2d1 dwrite windowscodecs ole32 uuid bcrypt dwmapi comdlg32 shell32 user32 gdi32)
  if(MINGW)
    target_link_options(${target} PRIVATE -static -municode)
  endif()
endforeach()
